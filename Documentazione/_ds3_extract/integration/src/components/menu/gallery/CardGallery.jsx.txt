// ============================================================
// CardGallery.jsx — Schermata "Galleria delle Carte"
// ============================================================
// Terza schermata "Galleria" del design system, gemella di
// GalleryCinematic (campi di battaglia) ma con chrome più semplice:
// un indice/griglia da sfogliare, non un carosello cinematic — è pura
// consultazione, senza fasi di intro/conferma.
//
// Mostra le carte nel layout ufficiale sashNameHud (CardReworkP4).
//
// ATTENZIONE — dati stub: cardGalleryDemoData.js è un campione di 12
// carte con nome/POT/DAN INVENTATI. Prima di integrare, sostituisci
// buildCards() con un adapter sui TUOI dati reali (src/data/cards.js
// → ARMY_SETS / le tue ~120 carte con nome/POT/DAN/abilità veri),
// mappando { id, army, name, pot, dan, ability, img } dal tuo schema.
// Vedi CardGallery.README.md.
//
// Props:
//   onBack()   torna al menu
// ============================================================

import React, { useState, useMemo } from 'react';
import { ARMY_COLORS, ARMY_BONUSES } from '../../../data/armies.js';
import { ARMY_LORE } from '../cosmic/armyLore.js';
import { CARD_GALLERY_CARDS } from './cardGalleryDemoData.js';

function buildCards(assetBase) {
  const base = assetBase || '/card-images/'; // Vite serve public/card-images/ come /card-images/
  return CARD_GALLERY_CARDS.map((c) => ({
    ...c,
    accent: (ARMY_COLORS[c.army] || {}).accent || '#94a3b8',
    glyph: (ARMY_LORE[c.army] || {}).glyph || '◈',
    bonus: ARMY_BONUSES[c.army] || '—',
    img: `${base}${c.id}.png`,
    league: Math.max(1, Math.min(5, Math.round((c.pot + c.dan) / 4))),
  }));
}

function armyList(cards) {
  const seen = [];
  cards.forEach((c) => { if (!seen.includes(c.army)) seen.push(c.army); });
  return seen;
}

export default function CardGallery({ onBack, assetBase }) {
  const CARDS = useMemo(() => buildCards(assetBase), [assetBase]);
  const ARMIES = useMemo(() => armyList(CARDS), [CARDS]);
  const [filter, setFilter] = useState(null); // null = tutte
  const [active, setActive] = useState(null); // carta in lightbox

  const shown = filter ? CARDS.filter((c) => c.army === filter) : CARDS;
  const headAccent = filter ? (ARMY_COLORS[filter] || {}).accent || '#f5f3eb' : '#f5f3eb';

  return (
    <div className="cgl" style={{ '--accent': headAccent }}>
      <div className="cgl-bg">
        <div className="cgl-bg-glow" />
        <div className="cgl-stars" />
        <div className="cgl-diag" />
      </div>

      <header className="cgl-top">
        <button className="cgl-back" onClick={onBack}>
          <span className="ar">←</span><span className="lbl">MENU</span>
        </button>
        <div className="cgl-title-block">
          <div className="cgl-eyebrow">ARCHIVIO DI GUERRA</div>
          <h1 className="cgl-title">GALLERIA DELLE CARTE</h1>
          <div className="cgl-sub">{shown.length} DI 120 CARTE MOSTRATE · layout ufficiale sashNameHud</div>
        </div>
        <div className="cgl-counter">
          <div className="ft-lbl">MOSTRATE</div>
          <div className="ft-name">{String(shown.length).padStart(2, '0')}<span className="sep">/</span>120</div>
        </div>
      </header>

      {/* Filtri armata */}
      <div className="cgl-filters">
        <button className={`cgl-chip ${!filter ? 'on' : ''}`} style={{ '--c': '#f5f3eb' }} onClick={() => setFilter(null)}>
          <span className="dot" /><span>TUTTE</span>
        </button>
        {ARMIES.map((a) => (
          <button key={a} className={`cgl-chip ${filter === a ? 'on' : ''}`} style={{ '--c': (ARMY_COLORS[a] || {}).accent || '#94a3b8' }} onClick={() => setFilter(a)}>
            <span className="dot" /><span>{a}</span>
          </button>
        ))}
      </div>

      {/* Griglia carte */}
      <div className="cgl-grid">
        {shown.map((c) => (
          <CardTile key={c.id} card={c} onClick={() => setActive(c)} />
        ))}
      </div>

      {/* Lightbox */}
      {active && <Lightbox card={active} onClose={() => setActive(null)} />}

      <div className="cgl-scanlines" />
      <CardGalleryStyles />
    </div>
  );
}

// ============================================================
// TILE — sashNameHud in miniatura
// ============================================================
function CardTile({ card, onClick }) {
  return (
    <button className="cgl-tile" style={{ '--accent': card.accent }} onClick={onClick}>
      <div className="cgl-card">
        <div className="cgl-art"><img src={card.img} alt={card.name} draggable={false} /></div>
        <div className="cgl-topgrad" />
        <FasciaSvg accent={card.accent} />
        <div className="cgl-fascia-row">
          <div className="cgl-circ pot"><div className="lbl">POT</div><div className="num">{card.pot}</div></div>
          <div className="cgl-nm">{card.name}</div>
          <div className="cgl-circ dan"><div className="lbl">DAN</div><div className="num">{card.dan}</div></div>
        </div>
        <div className="cgl-armybar" style={{ background: card.accent }}>
          <span className="an">{card.army}</span>
          <span className="lg">L{card.league}</span>
        </div>
      </div>
    </button>
  );
}

function FasciaSvg({ accent }) {
  const marks = [
    'M -3 -3 L 3 -3 L 0 3 Z', 'M -3 0 L 0 -3 L 3 0 L 0 3 Z', 'M -3 -3 L 3 3 M -3 3 L 3 -3',
    'M -3 0 L 3 0 M 0 -3 L 0 3', 'M -3 -2 L 3 -2 M -3 2 L 3 2', 'M 0 -3 L 3 0 L 0 3 L -3 0 Z',
  ];
  return (
    <svg className="cgl-fascia-svg" width="230" height="52" viewBox="0 0 220 52" preserveAspectRatio="none">
      <defs>
        <linearGradient id="cgl-g" x1="0" y1="0" x2="1" y2="0">
          <stop offset="0%" stopColor={accent} stopOpacity="0.4" /><stop offset="20%" stopColor={accent} />
          <stop offset="80%" stopColor={accent} /><stop offset="100%" stopColor={accent} stopOpacity="0.4" />
        </linearGradient>
      </defs>
      <rect x="0" y="15" width="220" height="22" fill="url(#cgl-g)" />
      <line x1="0" y1="14" x2="220" y2="14" stroke={accent} strokeWidth="0.8" />
      <line x1="0" y1="38" x2="220" y2="38" stroke={accent} strokeWidth="0.8" />
      <g stroke={accent} strokeWidth="0.9" fill="none" opacity="0.9">
        {[14, 41, 69, 96, 124, 151].map((x, i) => (
          <path key={`t${i}`} d={marks[i % marks.length]} transform={`translate(${x} 7)`} />
        ))}
        {[14, 41, 69, 96, 124, 151].map((x, i) => (
          <path key={`b${i}`} d={marks[(i + 3) % marks.length]} transform={`translate(${x} 45)`} />
        ))}
      </g>
      <rect x="0" y="24" width="220" height="2" fill="rgba(248,250,252,0.2)" />
    </svg>
  );
}

        <button className="cgl-lb-close" onClick={onClose}>✕</button>
        <div className="cgl-lb-card">
          <div className="cgl-art"><img src={card.img} alt={card.name} draggable={false} /></div>
          <div className="cgl-topgrad" />
          <FasciaSvg accent={card.accent} />
          <div className="cgl-fascia-row">
            <div className="cgl-circ pot"><div className="lbl">POT</div><div className="num">{card.pot}</div></div>
            <div className="cgl-nm">{card.name}</div>
            <div className="cgl-circ dan"><div className="lbl">DAN</div><div className="num">{card.dan}</div></div>
          </div>
          <div className="cgl-footer">
            <div className="frow"><span className="fkey pot"><span className="ic">★</span>POTERE</span><span className="fval">{card.ability}</span></div>
            <div className="fdiv" />
            <div className="frow"><span className="fkey bon"><span className="ic">✠</span>BONUS ARMATA</span><span className="fval">{card.bonus}</span></div>
          </div>
          <div className="cgl-armybar" style={{ background: card.accent }}>
            <span className="an">{card.army}</span>
            <span className="lg">L{card.league}</span>
          </div>
        </div>
      </div>
    </div>
  );
}
